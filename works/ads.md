<script>
(function(){
  var box=document.getElementById('lightbox');
  var img=document.getElementById('lightbox-img');
  function openBox(src,alt){ img.src=src; img.alt=alt||''; box.hidden=false; }
  document.getElementById('lightbox-close').addEventListener('click',function(){box.hidden=true;});
  box.addEventListener('click',function(){box.hidden=true;});
  document.querySelectorAll('[data-carousel]').forEach(function(car){
    var figs=car.querySelectorAll('figure');
    var count=car.querySelector('.car-count');
    var prev=car.querySelector('.car-prev');
    var next=car.querySelector('.car-next');
    var idx=0, timer=null;
    if(figs.length<2){ if(prev) prev.style.display='none'; if(next) next.style.display='none'; if(count) count.style.display='none'; }
    function show(n){
      idx=(n+figs.length)%figs.length;
      figs.forEach(function(f,i){ f.classList.toggle('active',i===idx); });
      if(count) count.textContent=(idx+1)+' / '+figs.length;
    }
    function start(){
      if(window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
      if(figs.length<2) return;
      stop();
      timer=setInterval(function(){ show(idx+1); },4000);
    }
    function stop(){ if(timer){ clearInterval(timer); timer=null; } }
    if(prev) prev.addEventListener('click',function(){ stop(); show(idx-1); start(); });
    if(next) next.addEventListener('click',function(){ stop(); show(idx+1); start(); });
    var x0=null;
    car.addEventListener('touchstart',function(e){ x0=e.touches[0].clientX; },{passive:true});
    car.addEventListener('touchend',function(e){
      if(x0===null) return;
      var dx=e.changedTouches[0].clientX-x0;
      if(Math.abs(dx)>40){ stop(); show(idx+(dx<0?1:-1)); start(); }
      x0=null;
    },{passive:true});
    car.addEventListener('mouseenter',stop);
    car.addEventListener('mouseleave',start);
    car.querySelectorAll('img').forEach(function(im){
      im.addEventListener('click',function(){ openBox(im.src,im.alt); });
    });
    show(0); start();
  });
  document.querySelectorAll('.ads-grid img').forEach(function(im){
    im.addEventListener('click',function(){ openBox(im.src,im.alt); });
  });
})();
</script>
